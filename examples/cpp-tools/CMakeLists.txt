cmake_minimum_required(VERSION 4.2)
project(OriginalRanges VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CT_STEP 1)
configure_file(include/original_step.h.in generated/original_step.h @ONLY)
add_library(original_ranges STATIC src/range.c src/range.cpp)
target_include_directories(original_ranges PRIVATE include "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_executable(original_c_boundary tests/c_boundary.c)
target_include_directories(original_c_boundary PRIVATE include "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(original_c_boundary PRIVATE original_ranges)
add_executable(original_cpp_boundary tests/cpp_boundary.cpp)
target_include_directories(original_cpp_boundary PRIVATE include "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(original_cpp_boundary PRIVATE original_ranges)
enable_testing()
add_test(NAME OriginalCBoundaries COMMAND original_c_boundary)
add_test(NAME OriginalCppBoundaries COMMAND original_cpp_boundary)
set_tests_properties(OriginalCBoundaries OriginalCppBoundaries PROPERTIES TIMEOUT 10)
